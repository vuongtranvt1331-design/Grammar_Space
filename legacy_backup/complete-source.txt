    function validateFirebaseSettings(value) {
      if (!value || typeof value !== 'object' || Array.isArray(value)) throw new Error('Vui lòng nhập cấu hình dưới dạng một đối tượng JSON.');
      const config = {};
      ['apiKey', 'authDomain', 'projectId', 'appId', 'storageBucket', 'messagingSenderId', 'measurementId'].forEach(key => {
        if (typeof value[key] === 'string' && value[key].trim()) config[key] = value[key].trim();
      });
      if (!['apiKey', 'authDomain', 'projectId'].every(key => config[key] && !config[key].startsWith('YOUR_'))) throw new Error('Cấu hình cần apiKey, authDomain và projectId của dự án Firebase.');
      return config;
    }
    function readFirebaseSettings() {
      try { return validateFirebaseSettings(window.GRAMMARSPACE_FIREBASE_CONFIG || JSON.parse(localStorage.getItem(FIREBASE_SETTINGS_KEY) || '{}')); }
      catch { return {}; }
    }

export function withDeadline(promise, milliseconds) {
  let timer;
  const timeout = new Promise((_, reject) => { timer = setTimeout(() => reject(Object.assign(new Error('Request timed out'), { code: 'deadline-exceeded' })), milliseconds); });
  return Promise.race([promise, timeout]).finally(() => clearTimeout(timer));
}
function configKey(text) {
  let hash = 2166136261;
  for (let i = 0; i < text.length; i++) hash = Math.imul(hash ^ text.charCodeAt(i), 16777619);
  return (hash >>> 0).toString(36);
}
export async function getUpdatesDatabase() {
  if (!window.firebase?.initializeApp || !window.firebase?.firestore) throw Object.assign(new Error('Firebase SDK unavailable'), { code: 'sdk-unavailable' });
  if (!database) {
    const name = 'grammarspace-student-' + configKey(JSON.stringify(config));
    const app = window.firebase.apps.find(item => item.name === name) || window.firebase.initializeApp(config, name);
    database = app.firestore();
  }
  return database;
}
export async function fetchNewUpdates() {
  const db = await getUpdatesDatabase();
  return withDeadline(db.collection('new_updates').get({ source: 'server' }), 15000);
}
export async function streamNewUpdates(onSnapshot, onError) {
  const db = await getUpdatesDatabase();
  return db.collection('new_updates').onSnapshot(onSnapshot, onError);
}

    function normalizeStoredProgress(item, total, now) {
      let score = null, completedAt = null;
      if (total > 0 && typeof item.score === 'number' && Number.isFinite(item.score) && item.score >= 0 && item.score <= 100) {
        score = item.score;
      } else if (item.score && typeof item.score === 'object') {
        // Migrate the earlier { correct, total, accuracy, completedAt } score format.
        const legacy = item.score;
        if (total > 0 && Number.isInteger(legacy.correct) && legacy.correct >= 0 && legacy.correct <= total && legacy.total === total) {
          score = Math.round(legacy.correct / total * 100);
          completedAt = typeof legacy.completedAt === 'string' ? Date.parse(legacy.completedAt) : null;
        }

    function getLessonProgress(lesson, now = Date.now()) {

      const daysPassed = Math.max(0, Math.floor((now - lesson.lastCompletedDate) / DAY_IN_MS));
      const currentScore = Math.max(0, lesson.score - daysPassed * DAILY_SCORE_DECAY);

    function loadLibrary(notice) {

        });
        
        if (needsMigration) {
          try {
            // Migrate just the progress fields, preserving other stored lesson metadata.
            const migrated = saved.map((item, index) => ({ ...item, score: loaded[index].score, lastCompletedDate: loaded[index].lastCompletedDate }));
            localStorage.setItem(STORAGE_KEY, JSON.stringify(migrated));
          } catch { notice('Your previous scores are available in this tab, but the updated history could not be saved.'); }
        }
        return { lessons: loaded, writable: true };
      } catch {
        
        notice('Saved lessons could not be loaded. You can still use this tab; existing stored data will be preserved.');
        return { lessons: null, writable: false };
      }
    }

        const aliases = { gap: 'gap-fill', 'fill-in-the-blank': 'gap-fill', mcq: 'multiple-choice', 'multiple_choice': 'multiple-choice', scramble: 'unscramble' };
        const declaredType = exercise.type;
        if (declaredType !== undefined && typeof declaredType !== 'string') fail(label, 'type phải là một chuỗi.');
        const type = aliases[declaredType] || declaredType || (own(exercise, 'options') || hasChoices ? 'multiple-choice' : own(exercise, 'words') || hasOrder ? 'unscramble' : 'gap-fill');
        if (![ 'gap-fill', 'multiple-choice', 'unscramble' ].includes(type)) fail(label, `loại câu hỏi “${type}” chưa được hỗ trợ.`);
        if ((hasGaps && type !== 'gap-fill') || (hasChoices && type !== 'multiple-choice') || (hasOrder && type !== 'unscramble')) fail(label, 'không trộn nhiều loại bài tập trong cùng một câu hỏi.');

        if (type === 'gap-fill') {
          const hints = [];
          const externalHint = position => Array.isArray(feedback) ? feedback[position] || genericFeedback : feedback;
          function transform(node) {
            if (node.nodeType === Node.ELEMENT_NODE && node.tagName === 'INPUT') {
              if ((node.getAttribute('type') || 'text').toLowerCase() !== 'text' || !(node.getAttribute('data-answer') || '').trim()) fail(label, 'mỗi ô trống cần input kiểu text và data-answer không rỗng.');
              hints.push(externalHint(hints.length));
              return;
            }
            if (node.nodeType !== Node.TEXT_NODE) { [...node.childNodes].forEach(transform); return; }
            const value = node.textContent;
            const pattern = /\{\{([^{}]*?)::([^{}]*?)\}\}/g;
            const matches = [...value.matchAll(pattern)];
            if (/\{\{|\}\}/.test(value.replace(pattern, ''))) fail(label, 'ô trống phải có dạng {{đáp án::giải thích}} và không bị ngắt bởi thẻ HTML.');
            const fragment = document.createDocumentFragment();
            let cursor = 0;
            matches.forEach(match => {
              const answer = match[1].trim();
              if (!answer) fail(label, 'đáp án trong {{…::…}} không được rỗng.');
              fragment.append(formattedFragment(value.slice(cursor, match.index)));
              const gap = document.createElement('input');
              gap.type = 'text'; gap.className = 'gap-input'; gap.dataset.answer = answer;
              fragment.append(gap);
              hints.push(match[2].trim() || externalHint(hints.length));
              cursor = match.index + match[0].length;
            });
            fragment.append(formattedFragment(value.slice(cursor)));
            node.replaceWith(fragment);
          }
          [...template.content.childNodes].forEach(transform);
          if (!hints.length) fail(label, 'thiếu câu hỏi: hãy thêm {{đáp án::giải thích}} hoặc input có data-answer.');
          if (Array.isArray(feedback) && feedback.length !== hints.length) fail(label, 'số phần tử feedback phải bằng số ô trống.');
          const cleaned = cleanHTML(htmlOf(template.content), true, index + 1);
          return { type, questionHTML: cleaned.html, feedback: hints };
        }

        if (type === 'multiple-choice') {
          let options = exercise.options, answer = exercise.answer, hint = singleFeedback(feedback);
          const marker = extractMarker(template, '((', '))', label);
          if (marker) {
            if (own(exercise, 'options') || own(exercise, 'answer')) fail(label, 'hãy dùng options/answer hoặc ký hiệu ((…)), không dùng cả hai.');
            const parsed = splitFeedback(marker.body);
            const choices = choiceValues(parsed.content, label);
            options = choices.options; answer = choices.answer;
            hint = parsed.feedback || hint;
            marker.replace(choices.prompt);
          } else if (hasChoices) fail(label, 'câu trắc nghiệm phải có đủ (( và )).');
          const values = validateChoices(options, answer, label);
          formatRemainingText(template.content);
          const prompt = cleanHTML(htmlOf(template.content));
          return { type, questionHTML: prompt.text ? prompt.html : choicePrompt, ...values, feedback: hint };
        }

        let words = exercise.words, hint = singleFeedback(feedback);
        const marker = extractMarker(template, '~~', '~~', label);
        if (marker) {
          if (own(exercise, 'words')) fail(label, 'hãy dùng words hoặc ký hiệu ~~…~~, không dùng cả hai.');
          const parsed = splitFeedback(marker.body);
          words = parsed.content.split('|').map(word => word.trim());
          hint = parsed.feedback || hint;
          marker.replace();
        }
        if (!Array.isArray(words) || !words.length || !words.every(word => typeof word === 'string' && word.trim())) fail(label, 'bài sắp xếp từ cần danh sách words không rỗng hoặc ~~ từ 1 | từ 2 ~~.');
        formatRemainingText(template.content);
        const prompt = cleanHTML(htmlOf(template.content));
        return { type, questionHTML: prompt.text ? prompt.html : orderPrompt, words: words.map(word => word.trim()), feedback: hint };
      }

      function theoryPages(html, label) {
        const template = inertHTML(html);
        formatRemainingText(template.content);
        const markers = [];
        textNodes(template.content).forEach(node => {
          for (const match of node.textContent.matchAll(/---\s*PAGE BREAK\s*---/g)) markers.push({ node, start: match.index, end: match.index + match[0].length });
        });
        let chunks = [];
        if (markers.length) {
          let startNode = template.content, startOffset = 0;
          markers.forEach(marker => {
            const range = document.createRange();
            range.setStart(startNode, startOffset); range.setEnd(marker.node, marker.start);
            chunks.push(htmlOfRange(range, template.content));
            startNode = marker.node; startOffset = marker.end;
          });
          const range = document.createRange();
          range.setStart(startNode, startOffset); range.setEnd(template.content, template.content.childNodes.length);
          chunks.push(htmlOfRange(range, template.content));
          if (chunks.some(chunk => !cleanHTML(chunk).text)) fail(label, 'không để trang lý thuyết trống trước hoặc sau --- PAGE BREAK ---.');
        } else {
          let container = template.content;
          // Authors often wrap the entire lesson in a single div. Treat its
          // immediate headings as logical boundaries while retaining the wrapper.
          while (true) {
            const significant = [...container.childNodes].filter(node => node.nodeType === Node.ELEMENT_NODE ? node.tagName !== 'BR' : node.nodeType === Node.TEXT_NODE && node.textContent.trim());
            if (significant.length !== 1 || significant[0].nodeType !== Node.ELEMENT_NODE || !['DIV', 'ARTICLE', 'MAIN', 'SECTION'].includes(significant[0].tagName)) break;
            container = significant[0];
          }
          let startNode = template.content, startOffset = 0;
          const addRange = (endNode, endOffset) => {
            const range = document.createRange();
            range.setStart(startNode, startOffset); range.setEnd(endNode, endOffset);
            const chunk = htmlOfRange(range, template.content);
            if (cleanHTML(chunk).text) chunks.push(chunk);
          };
          [...container.childNodes].forEach((node, index) => {
            if (node.nodeType !== Node.ELEMENT_NODE || !['H2', 'SECTION'].includes(node.tagName)) return;
            addRange(container, index);
            startNode = container; startOffset = index;
          });
          addRange(template.content, template.content.childNodes.length);
        }
        return chunks.map(chunk => {
          const clean = cleanHTML(chunk);
          const contents = inertHTML(clean.html);
          const heading = contents.content.querySelector('h1,h2,h3,h4');
          return { title: heading?.textContent.trim() || '', html: clean.html };
        });
      }

      let data = raw;
      if (typeof raw === 'string') {
        try { data = JSON.parse(raw.replace(/^\uFEFF/, '')); }
        catch { fail('JSON', 'không đọc được dữ liệu; hãy kiểm tra dấu ngoặc, dấu phẩy và dấu nháy kép.'); }
      }
      if (!isObject(data)) fail('Bài học', 'dữ liệu phải là một đối tượng JSON.');
      if (data.title !== undefined && typeof data.title !== 'string') fail('Bài học', 'title phải là một chuỗi.');
      if (typeof fallbackTitle !== 'string') fail('Bài học', 'tiêu đề dự phòng phải là một chuỗi.');
      const title = (data.title || '').trim() || fallbackTitle.trim();
      if (!title) fail('Bài học', 'thiếu tiêu đề; hãy thêm title.');
      const parts = [];
      function appendPart(part, position) {
        const label = `Phần ${position}`;
        if (typeof part === 'string') part = { lessonHTML: part };
        if (!isObject(part)) fail(label, 'mỗi phần phải là một đối tượng hoặc chuỗi HTML.');
        if (part.title !== undefined && typeof part.title !== 'string') fail(label, 'title phải là một chuỗi.');
        const partTitle = (part.title || '').trim() || (position === '1' ? title : `Phần ${position}`);
        const collections = ['parts', 'pages', 'sections'].filter(key => own(part, key));
        if (collections.length > 1) fail(label, 'chỉ dùng một trong các mảng parts, pages hoặc sections.');
        const html = own(part, 'lessonHTML') ? part.lessonHTML : own(part, 'html') ? part.html : (typeof part.content === 'string' ? part.content : '');
        if (typeof html !== 'string') fail(label, 'lessonHTML phải là một chuỗi.');
        const exercises = own(part, 'exercises') ? part.exercises : own(part, 'questions') ? part.questions : [];
        if (!Array.isArray(exercises)) fail(label, 'exercises phải là một mảng.');
        const pages = theoryPages(html, label);
        const questions = exercises.map((exercise, index) => parseQuestion(exercise, index, label));
        const firstPartIndex = parts.length;
        pages.forEach((page, index) => parts.push({ title: page.title || (pages.length > 1 ? `${partTitle} · ${index + 1}` : partTitle), lessonHTML: page.html, exercises: [] }));
        if (collections.length) {
          const children = part[collections[0]];
          if (!Array.isArray(children) || !children.length) fail(label, `${collections[0]} phải là một mảng không rỗng.`);
          children.forEach((child, index) => appendPart(child, `${position}.${index + 1}`));
        }
        for (let index = 0; index < questions.length; index += 10) {
          const chunk = questions.slice(index, index + 10);
          const last = parts[parts.length - 1];
          if (index === 0 && pages.length && !collections.length && last && !last.exercises.length) last.exercises = chunk;
          else parts.push({ title: questions.length > 10 ? `${partTitle} · Luyện tập ${Math.floor(index / 10) + 1}` : partTitle, lessonHTML: '', exercises: chunk });
        }
        if (parts.length === firstPartIndex) fail(label, 'thiếu nội dung lý thuyết hoặc câu hỏi.');
      }
      appendPart(data, '1');
      // Every HTML fragment above has already passed through cleanHTML. Keeping
      // the shared normalizer here makes persisted and newly imported data agree.
      return normalizeDocument(title, parts);
    }

import { createScoreRecord } from './storage-decay.js';

export function createNotebookUI({ byId, student, book, lessonProgress, reducedMotion, getActiveLesson, setActiveLesson, getLibrary, commitLesson, renderLibrary, showView }) {
    let pages = [], questions = [], practiceParts = [], currentPage = 0, turnState = null, turnTimer = 0;
    function measurePageHeight(page) {
      // Measure natural content without tying it to the book's existing height.
      // All visible sheets still occupy exactly 100% of the book after measuring.
      book.classList.add('is-measuring');
      const height = Math.ceil(page.face.offsetHeight);
      book.classList.remove('is-measuring');
      return height;
    }
    function syncBookHeight() {
      if (student.hidden || turnState || !pages[currentPage]) return;
      const height = measurePageHeight(pages[currentPage]);
      if (height && book.style.height !== height + 'px') book.style.height = height + 'px';
    }
    const pageObserver = new ResizeObserver(syncBookHeight);
    function cancelPageTurn() {
      clearTimeout(turnTimer); turnState = null; book.inert = false; book.classList.remove('is-turning'); book.removeAttribute('aria-busy');
    }
    function updatePageState(focus = false) {
      book.classList.add('is-resetting');
      pages.forEach((page, index) => {
        page.element.classList.toggle('flipped', index < currentPage);
        page.element.classList.toggle('is-visible', index === currentPage);
        page.element.classList.toggle('is-active', index === currentPage);
        page.element.inert = index !== currentPage;
        page.element.setAttribute('aria-hidden', String(index !== currentPage));
        page.previous.disabled = false; if (page.forward) page.forward.disabled = false;
      });
      if (pages[currentPage]) student.setAttribute('aria-labelledby', pages[currentPage].heading.id);
      void book.offsetWidth;
      book.classList.remove('is-resetting');
      syncBookHeight();
      if (focus && !student.hidden) pages[currentPage].heading.focus({ preventScroll: true });
    }
    function finishPageTurn() {
      if (!turnState) return;
      currentPage = turnState.to;
      cancelPageTurn(); updatePageState(true);
    }
    function turnPage(direction) {
      if (turnState || ![1, -1].includes(direction)) return;
      const target = direction > 0 ? currentPage + 1 : pages.map(page => page.element.classList.contains('flipped')).lastIndexOf(true);
      if (target < 0 || target >= pages.length) return;
      if (reducedMotion.matches) { currentPage = target; updatePageState(true); book.scrollIntoView({ block: 'start', behavior: 'auto' }); return; }
      const from = pages[currentPage].element, to = pages[target].element;
      const moving = direction > 0 ? from : to;
      turnState = { to: target, moving };
      const height = Math.max(measurePageHeight(pages[currentPage]), measurePageHeight(pages[target]));
      book.style.height = height + 'px';
      from.classList.add('is-visible'); to.classList.add('is-visible');
      pages.forEach(page => { page.previous.disabled = true; if (page.forward) page.forward.disabled = true; });
      book.classList.add('is-turning'); book.setAttribute('aria-busy', 'true'); book.inert = true;
      void moving.offsetWidth;
      if (direction > 0) moving.classList.add('flipped');
      else moving.classList.remove('flipped');
      book.scrollIntoView({ block: 'start', behavior: 'auto' });
      turnTimer = setTimeout(finishPageTurn, 950);
    }
    book.addEventListener('transitionend', event => { if (turnState && event.target === turnState.moving && event.propertyName === 'transform') finishPageTurn(); });
    book.addEventListener('transitioncancel', event => { if (turnState && event.target === turnState.moving && event.propertyName === 'transform') finishPageTurn(); });
    window.addEventListener('resize', syncBookHeight);
    document.fonts.ready.then(syncBookHeight);

    function shuffle(items, avoidOriginal = false) {
      const output = [...items];
      for (let i = output.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [output[i], output[j]] = [output[j], output[i]]; }
      if (avoidOriginal && output.length > 1 && output.every((item, index) => item === items[index])) output.push(output.shift());
      return output;
    }
    function updateLessonProgress() {
      const activeLesson = getActiveLesson();
      if (!activeLesson) return;
      if (!questions.length) { lessonProgress.textContent = 'This lesson contains theory only.'; return; }
      const checked = questions.filter(question => question.checked).length;
      if (checked === questions.length) {
        const correct = questions.filter(question => question.isCorrect).length;
        const unanswered = questions.filter(question => question.isUnanswered).length;
        lessonProgress.textContent = `${unanswered ? 'Attempt checked' : 'Lesson complete'} · ${Math.round(correct / questions.length * 100)}% · ${correct} of ${questions.length} questions correct.${unanswered ? ` ${unanswered} unanswered.` : ''}`;
      } else {
        const saved = getLibrary().find(item => item.id === activeLesson.id)?.score;
        lessonProgress.textContent = `${typeof saved === 'number' ? 'Last attempt: ' + Math.round(saved) + '%. ' : ''}Use Check Answers on the final page to grade all ${questions.length} questions.`;
      }
    }
    function clearQuestionFeedback(record) {
      record.card.classList.remove('is-correct', 'is-incorrect', 'is-unanswered');
      record.hints.hidden = true; record.success.hidden = true; record.unansweredMessage.hidden = true;
      record.hints.querySelectorAll('li').forEach(hint => { hint.hidden = true; });
      record.inputs.forEach(input => {
        input.classList.remove('correct', 'incorrect', 'unanswered');
        input.removeAttribute('aria-invalid'); input.removeAttribute('aria-describedby');
      });
      record.control?.removeAttribute('aria-invalid'); record.control?.removeAttribute('aria-describedby');
    }
    function invalidateQuestion(record) {
      const wasChecked = record.checked;
      record.checked = false; record.isCorrect = null; record.isUnanswered = null;
      clearQuestionFeedback(record);
      if (wasChecked) { record.part.result.textContent = 'Answers changed. Use Check Answers on the final page to update your score.'; updateLessonProgress(); }
    }
    function makeHint(record, text, id) {
      const hint = document.createElement('li'); hint.id = id; hint.hidden = true; hint.textContent = text; record.hints.append(hint); return hint;
    }
    function renderQuestion(data, part, index) {
      const prefix = `p${part.index + 1}-q${index + 1}`;
      const card = document.createElement('li'); card.className = `exercise exercise-${data.type}`;
      const number = document.createElement('span'); number.className = 'question-number'; number.textContent = String(index + 1).padStart(2, '0'); number.setAttribute('aria-hidden', 'true');
      const body = document.createElement('div'); body.className = 'question';
      const prompt = document.createElement('div'); prompt.id = prefix + '-prompt'; prompt.innerHTML = data.questionHTML;
      const hints = document.createElement('ul'); hints.className = 'feedback-list'; hints.hidden = true;
      const success = document.createElement('p'); success.className = 'success-message'; success.textContent = '✓ Nicely done!'; success.hidden = true;
      const unansweredMessage = document.createElement('p'); unansweredMessage.id = prefix + '-unanswered'; unansweredMessage.className = 'unanswered-message'; unansweredMessage.lang = 'vi'; unansweredMessage.textContent = 'Vui lòng hoàn thành câu này'; unansweredMessage.hidden = true;
      const record = { data, part, card, hints, success, unansweredMessage, checked: false, isCorrect: null, isUnanswered: null, inputs: [] };
      if (data.type !== 'gap-fill') { const kind = document.createElement('span'); kind.className = 'question-kind'; kind.textContent = data.type === 'multiple-choice' ? 'Choose one answer' : 'Build the sentence'; body.append(kind); }
      body.append(prompt);
      if (data.type === 'gap-fill') {
        record.inputs = [...prompt.querySelectorAll('.gap-input')];
        record.inputs.forEach((input, gapIndex) => {
          input.id = prefix + '-gap-' + (gapIndex + 1); input.autocomplete = 'off'; input.spellcheck = false; input.placeholder = '…';
          input.setAttribute('aria-label', `Part ${part.index + 1}, question ${index + 1}, gap ${gapIndex + 1}`);
          input.style.setProperty('--gap-width', `${Math.max(12, Math.min(Array.from(input.dataset.answer).length + 3, 28))}ch`);
          const hint = makeHint(record, (record.inputs.length > 1 ? `Gap ${gapIndex + 1}: ` : '') + (data.feedback[gapIndex] || 'Review the lesson and try again.'), input.id + '-feedback');
          input.dataset.feedbackId = hint.id;
          input.addEventListener('input', () => { invalidateQuestion(record); previewQuestion(record); });
        });
      } else if (data.type === 'multiple-choice') {
        const options = document.createElement('div'); options.className = 'mc-options'; options.setAttribute('role', 'radiogroup'); options.setAttribute('aria-labelledby', prompt.id); record.control = options;
        record.hint = makeHint(record, data.feedback, prefix + '-feedback');
        const name = `mc_${prefix}_${Math.random().toString(36).slice(2, 10)}`;
        shuffle(data.options).forEach((text, optionIndex) => {
          const label = document.createElement('label'); label.className = 'option-label';
          const input = document.createElement('input'); input.type = 'radio'; input.name = name; input.className = 'radio-input'; input.value = text; input.id = prefix + '-option-' + optionIndex;
          const letter = document.createElement('span'); letter.className = 'option-letter color-' + (optionIndex % 4); letter.textContent = (optionIndex < 26 ? String.fromCharCode(65 + optionIndex) : optionIndex + 1) + '.'; letter.setAttribute('aria-hidden', 'true');
          const caption = document.createElement('span'); caption.textContent = text;
          label.append(input, letter, caption); options.append(label); record.inputs.push(input);
          input.addEventListener('change', () => { record.inputs.forEach(radio => radio.parentElement.classList.toggle('is-selected', radio.checked)); invalidateQuestion(record); previewQuestion(record); });
        });
        body.append(options);
      } else {
        const sentenceLabel = document.createElement('p'); sentenceLabel.className = 'scramble-label'; sentenceLabel.textContent = 'Your sentence';
        const zone = document.createElement('div'); zone.className = 'drop-zone'; zone.setAttribute('role', 'group'); zone.setAttribute('aria-label', `Your sentence for part ${part.index + 1}, question ${index + 1}`);
        const bankLabel = document.createElement('p'); bankLabel.className = 'scramble-label'; bankLabel.textContent = 'Word bank · click a word to move it';
        const bank = document.createElement('div'); bank.className = 'word-bank'; bank.setAttribute('role', 'group'); bank.setAttribute('aria-label', `Word bank for part ${part.index + 1}, question ${index + 1}`);
        const announcement = document.createElement('span'); announcement.className = 'sr-only'; announcement.setAttribute('role', 'status'); announcement.setAttribute('aria-live', 'polite');
        record.control = zone; record.zone = zone; record.bank = bank; record.announcement = announcement; record.chips = []; record.hint = makeHint(record, data.feedback, prefix + '-feedback');
        shuffle(data.words.map((word, tokenIndex) => ({ word, tokenIndex })), true).forEach(token => {
          const chip = document.createElement('button'); chip.type = 'button'; chip.className = 'unscramble-chip'; chip.textContent = token.word; chip.dataset.word = token.word; chip.dataset.tokenId = prefix + '-' + token.tokenIndex;
          chip.setAttribute('aria-pressed', 'false'); chip.setAttribute('aria-label', `${token.word}. Add to sentence`);
          chip.addEventListener('click', () => {
            const add = chip.parentElement === bank; (add ? zone : bank).append(chip);
            chip.setAttribute('aria-pressed', String(add)); chip.setAttribute('aria-label', `${token.word}. ${add ? 'Return to word bank' : 'Add to sentence'}`);
            invalidateQuestion(record); previewQuestion(record); announcement.textContent = `${token.word} ${add ? 'added to your sentence' : 'returned to the word bank'}.`; chip.focus({ preventScroll: true });
          });
          record.chips.push(chip); bank.append(chip);
        });
        body.append(sentenceLabel, zone, bankLabel, bank, announcement);
      }
      card.append(number, body, hints, unansweredMessage, success); part.list.append(card); part.questions.push(record); questions.push(record);
    }
    const normalizeAnswer = value => value.normalize('NFC').trim().toLowerCase();
    function checkQuestion(record, preview = false) {
        // Clear the previous attempt before deciding whether any feedback may be shown.
        clearQuestionFeedback(record);
        const words = record.zone ? [...record.zone.querySelectorAll('.unscramble-chip')].map(chip => chip.dataset.word) : [];
        const unanswered = record.data.type === 'gap-fill'
          ? record.inputs.some(input => input.value.trim() === '')
          : record.data.type === 'multiple-choice'
            ? !record.inputs.some(input => input.checked)
            : words.length !== record.data.words.length;
        record.checked = !preview; record.isCorrect = false; record.isUnanswered = unanswered;
        if (unanswered && preview) return;
        if (unanswered) {
          // An incomplete question earns zero, but never exposes answer-specific hints.
          record.card.classList.add('is-unanswered'); record.unansweredMessage.hidden = false;
          if (record.data.type === 'gap-fill') {
            record.inputs.forEach(input => {
              if (input.value.trim() !== '') return;
              input.classList.add('unanswered'); input.setAttribute('aria-describedby', record.unansweredMessage.id);
            });
          } else record.control.setAttribute('aria-describedby', record.unansweredMessage.id);
          return;
        }
        let correct = true;
        if (record.data.type === 'gap-fill') {
          record.inputs.forEach(input => {
            const right = normalizeAnswer(input.value) === normalizeAnswer(input.dataset.answer), hint = byId(input.dataset.feedbackId);
            input.classList.toggle('correct', right); input.classList.toggle('incorrect', !right); input.setAttribute('aria-invalid', String(!right)); hint.hidden = right;
            if (right) input.removeAttribute('aria-describedby'); else { correct = false; input.setAttribute('aria-describedby', hint.id); }
          });
        } else {
          if (record.data.type === 'multiple-choice') correct = record.inputs.some(input => input.checked && input.value === record.data.answer);
          else correct = words.join(' ') === record.data.words.join(' ');
          record.hint.hidden = correct; record.control.setAttribute('aria-invalid', String(!correct));
          if (correct) record.control.removeAttribute('aria-describedby'); else record.control.setAttribute('aria-describedby', record.hint.id);
        }
        record.checked = !preview; record.isCorrect = correct;
        record.card.classList.toggle('is-correct', correct); record.card.classList.toggle('is-incorrect', !correct); record.hints.hidden = correct; record.success.hidden = !correct;

    function loadNewUpdates() {
      if (updatesRequest) return updatesRequest;
      if (!firebaseIsConfigured()) {

        renderSourcePanels(); return Promise.resolve();
      }
      updatesState.status = 'loading'; updatesState.message = ''; renderSourcePanels();
      updatesRequest = (async () => {
        try {
          // Read only: no uploads, scores, or personal renames are written to Firebase.
          const snapshot = await fetchNewUpdates();
          const lessons = []; let skipped = 0;
          snapshot.forEach(document => {
            try { lessons.push(normalizeCatalogLesson(document.data(), 'firebase', document.id)); }
            catch { skipped++; }
          });
          updatesState.lessons = lessons; updatesState.skipped = skipped; updatesState.status = 'ready';
          updatesState.message = skipped ? `${skipped} bài chưa hiển thị vì định dạng không hợp lệ.` : '';
        } catch (error) {
          updatesState.status = 'error';
          updatesState.message = updatesErrorMessage(error) + (updatesState.lessons.length ? ' Các bài đã tải trước đó vẫn ở bên dưới.' : '');
        } finally {
          updatesRequest = null;
          // An in-flight response never navigates away from the student's current screen.
          refreshLibrary(); if (!editingLessonId) renderLibrary(); else renderSourceStatus();
        }
      })();
      return updatesRequest;
    }
